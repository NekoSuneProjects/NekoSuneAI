# ─────────────────────────────────────────────────────────────────────────────
# NekoSuneAI — BASE requirements (text chat + headless web UI).
#
# This set is intentionally small, pure-Python where possible, and installs on
# every platform NekoSuneAI targets, including aarch64 / Raspberry Pi 5. It is
# all you need for the terminal chat loop (`python app.py`) and the
# browser-based web UI (`python app.py --web`).
#
# Voice (speech-in / speech-out / singing) and the native desktop GUI are
# optional add-ons — install them only if you want those features:
#   pip install -r requirements-voice.txt   # mic, STT, TTS/XTTS, embeddings
#   pip install -r requirements-gui.txt     # native pywebview desktop window
# ─────────────────────────────────────────────────────────────────────────────

python-dotenv>=1.0,<2
requests>=2.32,<3
duckduckgo-search>=6,<8
# openWakeWord's Raspberry Pi TFLite wheel is compiled against the NumPy 1.x
# C API. NumPy 2 raises `_ARRAY_API not found` before the listener can start.
numpy>=1.24,<2

# VRChat OSC integration (games/vrchat.py) — pure-Python, no heavy/compiled
# deps, so it ships in the base install rather than requirements-voice.txt.
python-osc>=1.9,<2

# VRChat friends system (games/vrchat_friends.py) — opt-in (off unless
# VRCHAT_FRIENDS_ENABLED + credentials are set in Settings), but lightweight
# and no known conflicts with anything else here, so it installs by default
# like the OSC integration above rather than needing a separate manual step.
vrchatapi>=1.1,<2
pyotp>=2.9,<3
websocket-client>=1.7,<2
paho-mqtt>=2.1,<3
vosk>=0.3.45,<0.4
